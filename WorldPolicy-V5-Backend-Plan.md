### eval.py — NEW V5 (Judge Gap #1)

**Purpose:** Runs the trained checkpoint through fixed benchmark scenarios and writes `data/eval_summary.json`. This is the proof artifact that separates "we trained a model" from "our model actually learned something."

**What it does:**
1. Loads `checkpoints/mappo_50k.pt`
2. Runs 10 evaluation episodes per policy (MAPPO, random, rule-based)
3. Uses 5 fixed random seeds (`seeds = [42, 7, 13, 99, 2026]`) — reproducible, stated in output
4. Records mean + std of: episode total reward, AID_DISPATCH frequency, arms_race_index at episode end
5. Also runs 3 fixed crisis scenarios (see below) for per-scenario comparison
6. Writes `data/eval_summary.json`
7. Calls `scripts/render_baselines.py` to generate `public/baseline_comparison.png`

```python
# eval.py

import torch, json, numpy as np
from world_env import WorldPolicyEnv
from mappo_agent import MAPPOPolicy

SEEDS       = [42, 7, 13, 99, 2026]
N_EPISODES  = 10   # per policy per seed → 50 rollouts total per policy
CHECKPOINT  = "checkpoints/mappo_50k.pt"

POLICIES = {
    "Random":     RandomPolicy(),
    "Rule-Based": RuleBasedPolicy(),   # NEUTRAL until disaster, then AID_DISPATCH
    "MAPPO":      MAPPOPolicy.load(CHECKPOINT)
}

# Fixed crisis scenarios (injected at episode start, otherwise env runs normally)
FIXED_SCENARIOS = [
    {"name": "early_disaster",    "inject_at_step": 10,  "disaster_severity": 0.9},
    {"name": "mid_war",           "inject_at_step": 50,  "crisis_type": "war"},
    {"name": "climate_emergency", "inject_at_step": 100, "crisis_type": "climate_emergency"}
]

def evaluate_policy(policy, n_episodes=N_EPISODES, seeds=SEEDS) -> dict:
    rewards = []
    aid_freqs = []
    arms_race_finals = []

    for seed in seeds:
        for ep in range(n_episodes // len(seeds)):
            env = WorldPolicyEnv()
            obs = env.reset(seed=seed + ep * 1000)
            total_reward = 0.0
            aid_count    = 0
            steps        = 0

            while True:
                actions = policy.act(obs)
                obs, rewards_step, terminated, truncated, info = env.step(actions)
                total_reward += sum(rewards_step.values())
                aid_count    += sum(1 for a in actions.values() if a == AID_DISPATCH)
                steps        += 1
                if terminated or truncated:
                    break

            rewards.append(total_reward)
            aid_freqs.append(aid_count / (steps * N_AGENTS))
            arms_race_finals.append(info.get("arms_race_index", 0.0))

    return {
        "mean_reward":         float(np.mean(rewards)),
        "std_reward":          float(np.std(rewards)),
        "mean_aid_freq":       float(np.mean(aid_freqs)),
        "mean_arms_race_final": float(np.mean(arms_race_finals)),
        "n_episodes":          len(rewards),
        "seeds":               seeds
    }

def run_eval():
    results = {}
    for name, policy in POLICIES.items():
        print(f"Evaluating {name}...")
        results[name] = evaluate_policy(policy)

    # Compute % improvement of MAPPO over rule-based
    mappo_mean  = results["MAPPO"]["mean_reward"]
    rule_mean   = results["Rule-Based"]["mean_reward"]
    results["mappo_improvement_pct"] = round(
        (mappo_mean - rule_mean) / abs(rule_mean) * 100, 1
    ) if rule_mean != 0 else None

    results["checkpoint"]    = CHECKPOINT
    results["eval_episodes"] = N_EPISODES
    results["generated_at"]  = datetime.utcnow().isoformat()

    with open("data/eval_summary.json", "w") as f:
        json.dump(results, f, indent=2)

    print("data/eval_summary.json written.")

    # Generate PNG artifact
    import subprocess
    subprocess.run(["python", "scripts/render_baselines.py"], check=True)
    print("public/baseline_comparison.png written.")

if __name__ == "__main__":
    run_eval()
```

### tests/test_emergent_detectors.py — NEW V5 (Judge Gap #3)

**Purpose:** Each detector gets one synthetic positive case and one synthetic negative case. If a judge asks "what actually fires cold_war_detected?" — you run these tests and show the output.

```python
# tests/test_emergent_detectors.py

import numpy as np, pytest
from analytics import (
    detect_cold_war,
    detect_arms_race_spiral,
    detect_free_riders
)

# ── cold_war_detected ──────────────────────────────────────────────────────────

class TestColdWarDetector:

    def _make_relations(self, cross_bloc: float) -> np.ndarray:
        """
        Bloc 0 = agents [0,1,2], Bloc 1 = agents [3,4,5].
        Within-bloc relations = +0.6. Cross-bloc = cross_bloc param.
        """
        R = np.full((6, 6), cross_bloc)
        for i in range(3):
            for j in range(3):
                if i != j:
                    R[i][j] = 0.6
        for i in range(3, 6):
            for j in range(3, 6):
                if i != j:
                    R[i][j] = 0.6
        np.fill_diagonal(R, 0.0)
        return R

    def test_positive_cold_war(self):
        """bloc_count=2, mean cross-bloc relation=-0.61 → DETECTED"""
        relations = self._make_relations(cross_bloc=-0.61)
        result = detect_cold_war(relations)
        assert result["detected"] is True, "Expected cold_war=True, got False"
        assert result["bloc_count"] == 2
        assert result["mean_cross_bloc_relation"] < -0.5

    def test_negative_cold_war_weak_separation(self):
        """Cross-bloc relation=-0.3 (above threshold) → NOT detected"""
        relations = self._make_relations(cross_bloc=-0.3)
        result = detect_cold_war(relations)
        assert result["detected"] is False, "Expected cold_war=False (cross-bloc not hostile enough)"

    def test_negative_cold_war_three_blocs(self):
        """Three blocs → bloc_count=3 → NOT detected (condition: bloc_count==2)"""
        R = np.full((6, 6), -0.6)
        np.fill_diagonal(R, 0.0)
        # agents 0,1 in bloc A; agents 2,3 in bloc B; agents 4,5 in bloc C
        R[0][1] = R[1][0] = 0.7
        R[2][3] = R[3][2] = 0.7
        R[4][5] = R[5][4] = 0.7
        result = detect_cold_war(R)
        assert result["detected"] is False, "3-bloc world should not trigger cold_war"

# ── arms_race_spiral ───────────────────────────────────────────────────────────

class TestArmsRaceSpiralDetector:

    def test_positive_arms_race(self):
        """All 6 agents have Δmilitary > 0 for 5 consecutive steps → DETECTED"""
        # military_history: shape (5, 6) — each row is military[6] at that step
        military_history = np.array([
            [0.30, 0.30, 0.30, 0.30, 0.30, 0.30],
            [0.35, 0.32, 0.31, 0.33, 0.36, 0.31],
            [0.40, 0.35, 0.33, 0.37, 0.42, 0.34],
            [0.45, 0.39, 0.36, 0.40, 0.47, 0.38],
            [0.50, 0.44, 0.40, 0.44, 0.53, 0.42],
        ])
        result = detect_arms_race_spiral(military_history)
        assert result["detected"] is True

    def test_negative_arms_race_one_peaceful(self):
        """5 agents increase military, 1 stays flat → NOT detected (requires ALL 6)"""
        military_history = np.array([
            [0.30, 0.30, 0.30, 0.30, 0.30, 0.50],
            [0.35, 0.32, 0.31, 0.33, 0.36, 0.50],   # agent 5 stays flat
            [0.40, 0.35, 0.33, 0.37, 0.42, 0.50],
            [0.45, 0.39, 0.36, 0.40, 0.47, 0.50],
            [0.50, 0.44, 0.40, 0.44, 0.53, 0.50],

    def test_negative_arms_race_too_short(self):
```

### docs/detector_examples.md — NEW V5 (Judge Gap #3)

```markdown
# Emergent Phenomena Detectors — Trigger Conditions & Examples

This document defines exactly what fires each detector badge in the UI.
It is the canonical reference for "what actually triggers cold_war_detected?"

## cold_war_detected

**Condition:**
```
bloc_count == 2
AND mean_cross_bloc_relation < -0.5
```

**How bloc_count is computed:**
- Build a weighted graph from the relations[6×6] matrix
- Only include edges where relations[i][j] > 0.3 (positive alliances only)
- Run `networkx.community.greedy_modularity_communities()` on this graph
- `bloc_count = len(communities)`

**Positive example (fires the badge):**
- Step 83 of scripted fallback
- Agents 0,1,2 (USA, EU, UK) have mean mutual relation = +0.62
- Agents 3,4,5 (China, Russia, India) have mean mutual relation = +0.58
- Cross-bloc mean relation = -0.61
- `bloc_count = 2` → cold_war_detected = True
- Logged trigger: `"bloc_count=2, cross_bloc_rel=-0.61"`

**Negative example (does NOT fire):**
- Step 40 of a typical episode
- Relations still near initial value (0.0) — no clear blocs
- `bloc_count = 1` (one large weakly-connected community)
- cold_war_detected = False

---

## arms_race_spiral

**Condition:**
```
For all i in range(6): Δmilitary[i] > 0
for 5 consecutive steps
```

**How it is computed:**
- Keep a rolling buffer `military_history` of shape `(window, 6)`
- `window = 5`
- Each step: shift buffer, append current `military[6]`
- Check: `all((military_history[t+1] - military_history[t]) > 0 for t in range(4))`

**Positive example:**
- Steps 60–65: all 6 agents choose MILITARY_DEPLOY or ARMS_RACE for 5 consecutive steps
- military[i] increases monotonically for all i
- arms_race_spiral = True
- Logged trigger: `"all Δmilitary>0 for 5 consecutive steps (steps 60–65)"`

**Negative example:**
- Steps 60–65: 5 agents increase military, agent 3 (India) chooses AID_DISPATCH
- military[3] decreases by 0.10 at step 62
- arms_race_spiral = False

---

## free_rider_count

**Condition (per agent i):**
```
climate[i] > 0.6      # high climate score — benefiting from collective pledges
AND pledge_rate[i] < 0.1  # low individual pledge frequency
```

**How pledge_rate[i] is computed:**
```
pledge_rate[i] = count(CLIMATE_PLEDGE actions by agent i) / total_steps_so_far
```

**Positive example:**
- Step 120: China (agent 2) has climate=0.68 (above 0.6)
- China has chosen CLIMATE_PLEDGE only 8 times in 120 steps = pledge_rate 0.067
- free_rider_count = 1, agent_indices = [2]
- Logged trigger: `"China: climate=0.68, pledge_rate=0.07"`

**Negative example:**
- Step 120: Same climate score, but China chose CLIMATE_PLEDGE 15 times
- pledge_rate = 0.125 (above 0.1 threshold)
- China NOT flagged as free rider
```

### training_meta.py — NEW V5 (Judge Gap #5)

**Purpose:** Writes `data/training_meta.json` at the end of `train.py`. This is what `GET /training-meta` serves, and what `TrainingFactsCard.tsx` displays. Proves to sponsor judges that a real PyTorch system ran.

```python
# training_meta.py

import json, hashlib, os, torch
from datetime import datetime

def compute_checkpoint_sha(path: str) -> str:
    """First 8 chars of SHA256 of checkpoint file."""
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            h.update(chunk)
    return h.hexdigest()[:8]

def count_parameters(model: torch.nn.Module) -> int:
    return sum(p.numel() for p in model.parameters() if p.requires_grad)

def write_training_meta(
    backbone,
    critic,
    actor_heads,
    total_steps: int,
    rollout_len: int,
    lr: float,
    checkpoint_path: str
):
    total_params = (
        count_parameters(backbone)
        + count_parameters(critic)
        + sum(count_parameters(h) for h in actor_heads)
    )

    meta = {
        "architecture": {
            "backbone":     "MLP 36→128→128",
            "actor_heads":  "6 × Linear(128→8)",
            "critic":       "MLP 216→256→256→1 (centralized)",
            "total_params": total_params,

        },

        },

    }
```
